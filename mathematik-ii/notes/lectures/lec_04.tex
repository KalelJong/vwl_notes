\section{Folgen und Reihen}
\begin{definition}
    [Folge] (Das n-Glied der Folge) ${\left\{a_n\right\}}_{n \in \mathbb{N}}$ ist eine Funktion a: $\mathbb{N} \rightarrow \mathbb{R}$. Eine Funktionsvorschrift, die Natürlichen Zahlen an Reelen Zahlen zuweist.
\end{definition}

\begin{definition}
    [Reihen] gehören zu einer entsprechenden Folge, wobei sie der n-Summe der Folgenglieder gleichen.
    \begin{equation}
        {\left\{\sum_{i=1}^{n} a_i\right\}}_{n \in \mathbb{N}}
    \end{equation}
\end{definition}

\subsection{Eigenschaften}

\begin{definition}
    Eine Folge ${\left\{a_n\right\}}_{n \in \mathbb{N}}$  heisst:
    \begin{itemize}
        \item \textbf{monoton Steigend}, wenn $a_{n+1} \ge a_n \forall n \in \mathbb{N}$
        \item \textbf{streng Steigend}  $a_{n+1} > a_n \forall n \in \mathbb{N}$
            \subitem $a_n = 3n$
        \item \textbf{monoton fallend}
        \item \textbf{streng fallend}
        \item \textbf{beschränkt}  wenn $\left\{a_n \mid n \in \mathbb{N} \right\}$ (\textit{Wertebereich}) oben und unten beschränkt ist.
            \subitem Bsp. $a_n = (-1)^n$  
    \end{itemize}
\end{definition}

\subsection{Grenzwerte}
\begin{definition}
    [Grenzwert [a] einer Folge] existiert wenn für jedes Gewählte Abstand $\epsilon$ von a es einen Folgeglied-m gibt, ab dem alle grösseren glieder (n>m) näher zu a sind als $\epsilon$
    \begin{equation}
        \forall \epsilon \exists \left\{a_m\right\}_{m\in\mathbb{R}} | n > m | \quad a_n - a \quad < \epsilon 
    \end{equation}
 
    Gegeben einen Grenzwert und epsilon kann die Ungleichung nach dem entsprechende m-te Glied gelöst werden
\end{definition}

\begin{definition}
    [\textit{unengentliche} Grenzwert] umgekehrt zum eigentlichen Grenzwert.  \footnote{Für jeder vermuteter Grenzwert und gegebene epsilon-umgebung, gibt es einen m-te Folgeglied ab dem alle weiter entfernt sind.}

    \begin{equation}
         \forall a \exists \left\{a_m\right\}_{m\in\mathbb{R}} | n > m | a_n > a
    \end{equation}
\end{definition}

\subsection{Konvergenz und Beschränktheit}
\begin{itemize}
    \item konvergente Folge => beschränkt
    \begin{itemize}
        \item Umkehrung nicht zwingend war
        \item $-1^n$ beschränkt aber divergent  
    \end{itemize}
    \item Begrenzte und monoton fallend oder steigend => konvergent   
\end{itemize}
\subsection{Harmonische Folge und Reihe}
Für diese Folge Art sollte man Grenzwert kennen.

\begin{definition}
    [Harmonische Folge] 
    \begin{equation}
        a_n = \frac{1}{n^c}
    \end{equation}
    $n \in \mathbb{N}$ und $c>0$
\end{definition}

Bei einer folge 1/n wird
bei lim inf+ die korrespondierende Reihe unendlich gross, weil 1 + 1/2 (halb) + 1/3 + 1/4 + 1/6 (halb)
Beide fälle folge, die nach 0 kovergieren aber ihre Reihen unterschiedlich sind, ein mal unednlich, ein mal Konvergenz.

\begin{theorem}
    Jede konvergente Folge ist beschränkt.
\end{theorem}

\subsubsection{Quetschsatz}

\begin{theorem}
    [Quetschsatz] wenn 3 Folgen gegeben wobei jedes Glied der Folgen a <= b <= c und a und c gleiche Grenzwerte haben, muss b zwingend denselben Grenzwert haben.
\end{theorem}

\subsection{Besondere Folgen und Reihen}
\begin{definition}
    [arithmetische Folge] sind Folgen, bei den die Differenz zweier aufeinander folgenden Glieder immer konstant (d) ist.
    \begin{equation}
        a_{n+1} - a_n = d
    \end{equation}
\end{definition}

\begin{definition}
    [arithmetische Reihe] 
    \begin{equation}
        s_n = \sum_{i=i}^{n} (a_i (i - 1) d) 
    \end{equation}
    \begin{itemize}
        \item Wenn d > 0 => divergiert $+ \inf$
        \item wenn d< 0 => divergent $- \inf$
        \item wenn d = 0 konvergiert konstante $a_1$
    \end{itemize}
\end{definition}

\begin{definition}
    [geometrische Folge] sind Folgen, bei das Verhältnis/Quotient zweier aufeinander folgenden Glieder immer konstant (q) und > 1 ist. Wenn q < 1 konvergiert gegen 0
    \begin{equation}
        \frac{a_{n+1}}{a_n} = q, q\neq 1    
    \end{equation}
\end{definition}


\begin{definition}
    [geometrische Reihe] wenn |q| < 1
    \begin{equation}
       \lim_{n \rightarrow \inf} a_1 \frac{1-q^n}{1-q} = \lim_{n \rightarrow \inf} \frac{a}{1+q}
    \end{equation}
    Weil q^n gegen 0 konvergiert wenn |q| < 1 also für $0 \le q \le 1$
\end{definition}
